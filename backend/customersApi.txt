const isFiniteNum = (v) => Number.isFinite(Number(v));
const toNumOrNull = (v) => (isFiniteNum(v) ? Number(v) : null);
const trimOrNull = (v) => (v ?? '').toString().trim() || null;
const onlyDigits = (v) => (v ?? '').toString().replace(/\D+/g, '') || null;

function asBoolOrNull(v) {
  if (typeof v === 'boolean') return v;
  if (v === 'true' || v === '1' || v === 1) return true;
  if (v === 'false' || v === '0' || v === 0) return false;
  return null;
}

function normalizeCustomerType(value) {
  const v = String(value ?? '')
    .trim()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toUpperCase();

  if (
    v === 'F' ||
    v === 'PF' ||
    v === 'FISICA' ||
    v === 'PESSOA FISICA'
  ) {
    return 'F';
  }

  return 'J';
}

const DELIVERY_FIELDS = [
  'customer_ad_street_type',
  'customer_ad_street_name',
  'customer_ad_street_number',
  'customer_ad_address_complement',
  'customer_ad_neighborhood',
  'customer_ad_city',
  'customer_ad_state',
  'customer_ad_active',
];

function hasDeliveryAddressPayload(input = {}) {
  return DELIVERY_FIELDS.some(
    (field) => Object.prototype.hasOwnProperty.call(input, field)
  );
}

function sanitizeDeliveryAddress(input = {}) {
  const out = {};

  for (const field of DELIVERY_FIELDS) {
    if (!Object.prototype.hasOwnProperty.call(input, field)) continue;

    if (field === 'customer_ad_active') {
      const active = asBoolOrNull(input[field]);

      if (active == null) {
        throw new Error('customer_ad_active must be boolean');
      }

      out[field] = active;
      continue;
    }

    let value = trimOrNull(input[field]);

    if (field === 'customer_ad_state' && value) {
      value = value.toUpperCase();

      if (!/^[A-Z]{2}$/.test(value)) {
        throw new Error('customer_ad_state must contain 2 letters');
      }
    }

    out[field] = value;
  }

  return out;
}

async function saveDeliveryAddress(client, customerId, payload = {}) {
  if (!hasDeliveryAddressPayload(payload)) return null;

  const data = sanitizeDeliveryAddress(payload);

  const linkResult = await client.query(
    `SELECT customer_ad_id
       FROM ${TB}
      WHERE customer_id = $1
      FOR UPDATE`,
    [customerId]
  );

  if (!linkResult.rows[0]) return null;

  let addressId = toNumOrNull(linkResult.rows[0].customer_ad_id);

  const columnMap = {
    customer_ad_street_type: 'customer_ad_street_type',
    customer_ad_street_name: 'customer_ad_street_name',
    customer_ad_street_number: 'customer_ad_street_number',
    customer_ad_address_complement: 'customer_ad_address_complement',
    customer_ad_neighborhood: 'customer_ad_neighborhood',
    customer_ad_city: 'customer_ad_city',
    customer_ad_state: 'customer_ad_state',
    customer_ad_active: 'active',
  };

  if (addressId) {
    const params = [];
    const sets = [];

    for (const [field, value] of Object.entries(data)) {
      params.push(value);
      sets.push(`"${columnMap[field]}" = $${params.length}`);
    }

    if (sets.length) {
      sets.push('"updated_at" = NOW()');
      params.push(addressId);

      const updated = await client.query(
        `UPDATE ${TB_DELIVERY}
            SET ${sets.join(', ')}
          WHERE customer_ad_id = $${params.length}
          RETURNING customer_ad_id`,
        params
      );

      if (!updated.rowCount) {
        addressId = null;
      }
    }
  }

  if (!addressId) {
    const columns = [];
    const values = [];
    const params = [];

    for (const [field, value] of Object.entries(data)) {
      columns.push(`"${columnMap[field]}"`);
      params.push(value);
      values.push(`$${params.length}`);
    }

    const insertSql = columns.length
      ? `INSERT INTO ${TB_DELIVERY}
           (${columns.join(', ')})
         VALUES
           (${values.join(', ')})
         RETURNING customer_ad_id`
      : `INSERT INTO ${TB_DELIVERY}
         DEFAULT VALUES
         RETURNING customer_ad_id`;

    const inserted = await client.query(insertSql, params);

    addressId = inserted.rows[0].customer_ad_id;

    await client.query(
      `UPDATE ${TB}
          SET customer_ad_id = $1,
              updated_at = NOW()
        WHERE customer_id = $2`,
      [addressId, customerId]
    );
  }

  return addressId;
}

/* ──────────────────────────────────────────────────────────
   IBGE helpers
────────────────────────────────────────────────────────── */

function cleanUf(v) {
  return String(v || '').trim().toUpperCase().slice(0, 2);
}

function cleanCity(v) {
  return String(v || '').trim();
}

function cleanIbgeCode(v) {
  const s = String(v ?? '')
    .replace(/\D+/g, '')
    .slice(0, 7);

  return s || null;
}

/**
 * UPSERT customersIbge (idempotente).
 * - codigo_ibge undefined => não altera código.
 * - codigo_ibge null/string => atualiza com COALESCE.
 */
async function upsertCustomersIbge({
  customer_id,
  uf,
  city,
  codigo_ibge,
  source = 'manual',
  confidence = 100,
  db = pool,
} = {}) {
  const cid = toNumOrNull(customer_id);
  const UF = cleanUf(uf);
  const CITY = cleanCity(city);

  if (!cid) return;
  if (!UF || UF.length !== 2) return;
  if (!CITY || CITY.length < 2) return;

  const codeProvided = codigo_ibge !== undefined;
  const CODE = codeProvided ? cleanIbgeCode(codigo_ibge) : null;

  const sqlNoCode = `
    INSERT INTO ${TB_IBGE}
      (customer_id, uf, city, codigo_ibge, source, confidence, updated_at)
    VALUES
      ($1, $2, $3, NULL, COALESCE(NULLIF($4,''), 'manual'), COALESCE($5, 100), now())
    ON CONFLICT (customer_id)
    DO UPDATE SET
      uf          = EXCLUDED.uf,
      city        = EXCLUDED.city,
      source      = EXCLUDED.source,
      confidence  = EXCLUDED.confidence,
      updated_at  = now();
  `;

  const sqlWithCode = `
    INSERT INTO ${TB_IBGE}
      (customer_id, uf, city, codigo_ibge, source, confidence, updated_at)
    VALUES
      ($1, $2, $3, $4, COALESCE(NULLIF($5,''), 'manual'), COALESCE($6, 100), now())
    ON CONFLICT (customer_id)
    DO UPDATE SET
      uf           = EXCLUDED.uf,
      city         = EXCLUDED.city,
      codigo_ibge  = COALESCE(EXCLUDED.codigo_ibge, ${TB_IBGE}.codigo_ibge),
      source       = EXCLUDED.source,
      confidence   = EXCLUDED.confidence,
      updated_at   = now();
  `;

  if (!codeProvided) {
    await db.query(
      sqlNoCode,
      [
        cid,
        UF,
        CITY,
        String(source || '').trim(),
        Number(confidence) || 100,
      ]
    );

    return;
  }

  await db.query(
    sqlWithCode,
    [
      cid,
      UF,
      CITY,
      CODE,
      String(source || '').trim(),
      Number(confidence) || 100,
    ]
  );
}

/* ──────────────────────────────────────────────────────────
   Sanitização — CREATE
────────────────────────────────────────────────────────── */

function sanitizeCreate(input = {}) {
  const out = {};

  for (const k of FIELDS) {
    let v = input[k];

    // ---------------------------------------------------------
    // Tipo de pessoa
    // ---------------------------------------------------------
    if (k === 'customer_type') {
      v = normalizeCustomerType(v);
    }

    // ---------------------------------------------------------
    // Strings
    // ---------------------------------------------------------
    if (
      [
        'trade_name',
        'legal_name',
        'classification',
        'salesperson',
        'contact_name',
        'street_type',
        'street_name',
        'neighborhood',
        'city',
        'state',
        'business_segment',
        'email',
        'postal_code',
        'street_number',
        'address_complement',
        'phone_number',
        'mobile_number',
        'asaas_id',
        'state_registration',
        'municipal_registration',
        'financial_contact_name',
        'financial_contact_email',
        'financial_contact_phone',
        'financial_contact_mobile',
        'billing_email',
        'invoice_email',
        'financial_contact_notes',
      ].includes(k)
    ) {
      v = trimOrNull(v);
    }

    // CPF / CNPJ
    if (k === 'tax_id') {
      v = onlyDigits(v);
    }

    // Segmento
    if (k === 'segment_id') {
      v = toNumOrNull(v);
    }

    // Ativo
    if (k === 'is_active') {
      v = asBoolOrNull(v);
    }

    out[k] = v ?? null;
  }

  if (!out.customer_type) {
    out.customer_type = 'J';
  }

  if (out.is_active == null) {
    out.is_active = true;
  }

  // Pessoa Física
  if (out.customer_type === 'F') {
    out.state_registration = null;
    out.municipal_registration = null;

    if (!out.legal_name && out.trade_name) {
      out.legal_name = out.trade_name;
    }

    if (!out.trade_name && out.legal_name) {
      out.trade_name = out.legal_name;
    }
  }

  if (out.state) {
    out.state = String(out.state)
      .trim()
      .toUpperCase()
      .slice(0, 2);
  }

  if (out.postal_code) {
    out.postal_code = onlyDigits(out.postal_code)?.slice(0, 8) ?? null;
  }

  return out;
}

/* ──────────────────────────────────────────────────────────
   Sanitização — UPDATE
────────────────────────────────────────────────────────── */

function sanitizeUpdate(input = {}) {
  const out = {};

  for (const k of Object.keys(input || {})) {
    if (!FIELDS.includes(k)) continue;

    let v = input[k];

    // Tipo de pessoa
    if (k === 'customer_type') {
      v = normalizeCustomerType(v);
    }

    // Strings
    if (
      [
        'trade_name',
        'legal_name',
        'classification',
        'salesperson',
        'contact_name',
        'street_type',
        'street_name',
        'neighborhood',
        'city',
        'state',
        'business_segment',
        'email',
        'postal_code',
        'street_number',
        'address_complement',
        'phone_number',
        'mobile_number',
        'asaas_id',
        'state_registration',
        'municipal_registration',
        'financial_contact_name',
        'financial_contact_email',
        'financial_contact_phone',
        'financial_contact_mobile',
        'billing_email',
        'invoice_email',
        'financial_contact_notes',
      ].includes(k)
    ) {
      v = trimOrNull(v);
    }

    if (k === 'tax_id') {
      v = onlyDigits(v);
    }

    if (k === 'segment_id') {
      v = toNumOrNull(v);
    }

    if (k === 'is_active') {
      v = asBoolOrNull(v);
    }

    if (k === 'state' && v) {
      v = String(v)
        .trim()
        .toUpperCase()
        .slice(0, 2);
    }

    if (k === 'postal_code' && v) {
      v = onlyDigits(v)?.slice(0, 8) ?? null;
    }

    out[k] = v;
  }

  // Pessoa Física
  if (out.customer_type === 'F') {
    out.state_registration = null;
    out.municipal_registration = null;

    if (
      Object.prototype.hasOwnProperty.call(out, 'trade_name') &&
      !out.legal_name
    ) {
      out.legal_name = out.trade_name;
    }

    if (
      Object.prototype.hasOwnProperty.call(out, 'legal_name') &&
      !out.trade_name
    ) {
      out.trade_name = out.legal_name;
    }
  }

  return out;
}

            ADD COLUMN created_at TIMESTAMP DEFAULT now();

        -- updated_at
        IF NOT EXISTS (
          SELECT 1
          FROM information_schema.columns
          WHERE table_schema = 'public'
            AND table_name = 'customers'
            AND column_name = 'updated_at'
        ) THEN
          ALTER TABLE public.customers
            ADD COLUMN updated_at TIMESTAMP DEFAULT now();
        END IF;
      END$$;
    `);

    // ---------------------------------------------------------------
    // 4) FK endereço de entrega
    // ---------------------------------------------------------------
    await client.query(`
      DO $$
      BEGIN
        IF NOT EXISTS (
          SELECT 1
          FROM information_schema.table_constraints
          WHERE constraint_schema = 'public'
            AND table_name = 'customers'
            AND constraint_name = 'fk_customers_delivery_address'
        ) THEN
          ALTER TABLE public.customers
            ADD CONSTRAINT fk_customers_delivery_address
            FOREIGN KEY (customer_ad_id)
            REFERENCES public."customersAddressDelivery" (customer_ad_id)
            ON UPDATE CASCADE
            ON DELETE SET NULL;
        END IF;
      END$$;
    `);

    // ---------------------------------------------------------------
    // 5) FK segmento
    // ---------------------------------------------------------------
    await client.query(`
      DO $$
      BEGIN
        IF EXISTS (
          SELECT 1
          FROM information_schema.tables
          WHERE table_schema = 'public'
            AND table_name = 'customerSegments'
        ) THEN
          IF NOT EXISTS (
            SELECT 1
            FROM information_schema.table_constraints
            WHERE constraint_schema = 'public'
              AND table_name = 'customers'
              AND constraint_name = 'fk_customers_segment_id'
          ) THEN
            ALTER TABLE public.customers
              ADD CONSTRAINT fk_customers_segment_id
              FOREIGN KEY (segment_id)
              REFERENCES public."customerSegments"(segment_id)
              ON UPDATE CASCADE
              ON DELETE SET NULL;
          END IF;
        END IF;
      END$$;
    `);

    // ---------------------------------------------------------------
    // 6) Índices
    // ---------------------------------------------------------------
    await client.query(`
      DO $$
      BEGIN
        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_tax_id_like'
        ) THEN
          CREATE INDEX idx_customers_tax_id_like
            ON public.customers (tax_id);
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_tax_id_unique'
        ) THEN
          CREATE UNIQUE INDEX idx_customers_tax_id_unique
            ON public.customers (tax_id)
            WHERE tax_id IS NOT NULL;
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_taxid_unique_digits'
        ) THEN
          CREATE UNIQUE INDEX idx_customers_taxid_unique_digits
            ON public.customers (
              regexp_replace(
                COALESCE(tax_id, '')::text,
                '\D+'::text,
                ''::text,
                'g'::text
              )
            )
            WHERE tax_id IS NOT NULL
              AND length(
                regexp_replace(
                  tax_id::text,
                  '\D+'::text,
                  ''::text,
                  'g'::text
                )
              ) > 0;
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_tax_id_digits'
        ) THEN
          CREATE INDEX idx_customers_tax_id_digits
            ON public.customers (
              regexp_replace(
                COALESCE(tax_id, '')::text,
                '\D+'::text,
                ''::text,
                'g'::text
              )
            );
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_trade_name'
        ) THEN
          CREATE INDEX idx_customers_trade_name
            ON public.customers (trade_name);
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_legal_name'
        ) THEN
          CREATE INDEX idx_customers_legal_name
            ON public.customers (lower(legal_name::text));
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_city'
        ) THEN
          CREATE INDEX idx_customers_city
            ON public.customers (city);
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_city_state'
        ) THEN
          CREATE INDEX idx_customers_city_state
            ON public.customers (lower(city::text), state);
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_email_idx'
        ) THEN
          CREATE INDEX idx_customers_email_idx
            ON public.customers (lower(trim(email)))
            WHERE email IS NOT NULL
              AND trim(email) <> '';
        END IF;

        IF NOT EXISTS (
          SELECT 1
          FROM pg_indexes
          WHERE schemaname = 'public'
            AND indexname = 'idx_customers_segment_id'
        ) THEN
          CREATE INDEX idx_customers_segment_id
            ON public.customers (segment_id);
        END IF;

        IF EXISTS (
          SELECT 1
          FROM pg_extension
          WHERE extname = 'pg_trgm'
        ) THEN
          IF NOT EXISTS (
            SELECT 1
            FROM pg_indexes
            WHERE schemaname = 'public'
              AND indexname = 'idx_customers_trade_name_trgm'
          ) THEN
            CREATE INDEX idx_customers_trade_name_trgm
              ON public.customers
              USING gin (trade_name gin_trgm_ops);
          END IF;
        END IF;
      END$$;
    `);

    // ---------------------------------------------------------------
    // 7) Sequência PK
    // ---------------------------------------------------------------
    const seqRes = await client.query(
      `SELECT pg_get_serial_sequence($1, 'customer_id') AS seq_name`,
      [TABLE_PLAIN]
    );

    let seqName = seqRes.rows[0]?.seq_name;

    if (!seqName) {
      seqName = 'public.customers_customer_id_seq';

      await client.query(
        `CREATE SEQUENCE IF NOT EXISTS ${seqName};`
      );

      await client.query(`
        ALTER TABLE public.customers
        ALTER COLUMN customer_id
        SET DEFAULT nextval('${seqName}');
      `);

      await client.query(
        `ALTER SEQUENCE ${seqName}
         OWNED BY public.customers.customer_id;`
      );
    } else {
      await client.query(
        `ALTER SEQUENCE ${seqName}
         OWNED BY public.customers.customer_id;`
      );

      await client.query(`
        ALTER TABLE public.customers
        ALTER COLUMN customer_id
        SET DEFAULT nextval('${seqName}');
      `);
    }

    await client.query(
      `
      SELECT setval(
        $1,
        COALESCE(
          (SELECT MAX(customer_id) FROM public.customers),
          0
        ) + 1,
        false
      )
      `,
      [seqName]
    );

    await client.query('COMMIT');

    return { ok: true };
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}

/**
 * CREATE
 */
export async function create(payload = {}) {
  const data = sanitizeCreate(payload);

  if (!data.trade_name) {
    throw new Error('trade_name is required');
  }

  const client = await pool.connect();

  try {
    await client.query('BEGIN');

    const cols = [];
    const vals = [];
    const params = [];

    for (const k of FIELDS) {
      if (data[k] !== undefined) {
        cols.push(`"${k}"`);
        params.push(data[k]);
        vals.push(`$${params.length}`);
      }
    }

    const { rows } = await client.query(
      `INSERT INTO ${TB}
        (${cols.join(', ')})
       VALUES
        (${vals.join(', ')})
       RETURNING *`,
      params
    );

    const created = rows[0];

    await saveDeliveryAddress(
      client,
      created.customer_id,
      payload
    );

    await upsertCustomersIbge({
      customer_id: created.customer_id,
      uf: created.state,
      city: created.city,
      codigo_ibge:
        payload?.ibge_city_code ??
        payload?.codigo_ibge,
      source:
        payload?.ibge_source ??
        'manual',
      confidence:
        payload?.ibge_confidence ??
        100,
      db: client,
    });

    await client.query('COMMIT');

    return await getById(created.customer_id);
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}

/**
 * UPDATE BY ID
 */
export async function updateById(id, payload = {}) {
  const cid = toNumOrNull(id);

  if (!cid) {
    throw new Error('invalid customer_id');
  }

  const incomingCodigoIbge =
    payload?.ibge_city_code ??
    payload?.codigo_ibge;

  const incomingSource =
    payload?.ibge_source;

  const incomingConfidence =
    payload?.ibge_confidence;

  const data = sanitizeUpdate(payload);
  const client = await pool.connect();

  try {
    await client.query('BEGIN');

    const currentResult = await client.query(
      `SELECT *
         FROM ${TB}
        WHERE customer_id = $1
        FOR UPDATE`,
      [cid]
    );

    if (!currentResult.rows[0]) {
      await client.query('ROLLBACK');
      return null;
    }

    let updatedBase = currentResult.rows[0];

    const sets = [];
    const params = [];

    for (const [k, v] of Object.entries(data)) {
      params.push(v);
      sets.push(`"${k}" = $${params.length}`);
    }

    if (sets.length) {
      sets.push('"updated_at" = NOW()');
      params.push(cid);

      const updatedResult = await client.query(
        `UPDATE ${TB}
            SET ${sets.join(', ')}
          WHERE customer_id = $${params.length}
          RETURNING *`,
        params
      );

      updatedBase = updatedResult.rows[0];
    }

    await saveDeliveryAddress(
      client,
      cid,
      payload
    );

    const shouldUpsertIbge =
      payload?.state !== undefined ||
      payload?.city !== undefined ||
      incomingCodigoIbge !== undefined;

    if (shouldUpsertIbge) {
      await upsertCustomersIbge({
        customer_id: cid,
        uf: updatedBase.state,
        city: updatedBase.city,
        codigo_ibge: incomingCodigoIbge,
        source: incomingSource ?? 'manual',
        confidence: incomingConfidence ?? 100,
        db: client,
      });
    }

    await client.query('COMMIT');

    return await getById(cid);
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();
  }
}

/**
 * GET BY ID
 */
export async function getById(id) {
  const cid = toNumOrNull(id);

  if (!cid) {
    throw new Error('invalid customer_id');
  }

  const sql = `
    SELECT
      c.*,

      -- IBGE
      ci.codigo_ibge      AS ibge_city_code,
      ci.source           AS ibge_source,
      ci.confidence       AS ibge_confidence,

      -- STATUS
      cs.status_type_id,
      cst.name_pt         AS status_name_pt,
      cst.internal_code   AS status_internal_code,
      cst.color_hex       AS status_color_hex,
      COALESCE(cst.name_pt, cst.internal_code) AS status_label,

      -- SEGMENTO
      seg.segment_code    AS segment_code,
      seg.segment_name    AS segment_name,

      -- SELLER
      sfc.seller_id       AS seller_id,
      COALESCE(s.nickname, s.name) AS seller_name,

      -- ENDEREÇO DE ENTREGA
      ad.customer_ad_street_type,
      ad.customer_ad_street_name,
      ad.customer_ad_street_number,
      ad.customer_ad_address_complement,
      ad.customer_ad_neighborhood,
      ad.customer_ad_city,
      ad.customer_ad_state,
      ad.active AS customer_ad_active

    FROM ${TB} c

    LEFT JOIN ${TB_IBGE} ci
      ON ci.customer_id = c.customer_id

    LEFT JOIN ${TB_DELIVERY} ad
      ON ad.customer_ad_id = c.customer_ad_id

    LEFT JOIN public."customerStatus" cs
      ON cs.customer_id = c.customer_id

    LEFT JOIN public."customerStatusTypes" cst
      ON cst.id = cs.status_type_id

    LEFT JOIN ${TB_SEG} seg
      ON seg.segment_id = c.segment_id

    LEFT JOIN LATERAL (
      SELECT sfc1.seller_id
      FROM public."SellerForCustomer" sfc1
      WHERE sfc1.customer_id = c.customer_id
        AND (sfc1.active IS NULL OR sfc1.active = TRUE)
      ORDER BY sfc1.created_at DESC
      LIMIT 1
    ) sfc ON TRUE

    LEFT JOIN public.sellers s
      ON s.id = sfc.seller_id

    WHERE c.customer_id = $1
    LIMIT 1;
  `;

  const { rows } = await pool.query(sql, [cid]);

  return rows[0]
    ? mapRow(rows[0])
    : null;
}

/**
 * DELETE
 */
export async function removeById(id, { hard = false } = {}) {
  const cid = toNumOrNull(id);

  if (!cid) {
    throw new Error('invalid customer_id');
  }

  if (hard) {
    const { rowCount } = await pool.query(
      `DELETE FROM ${TB}
        WHERE customer_id = $1`,
      [cid]
    );

    return {
      ok: rowCount > 0,
      hard: true,
    };
  }

  const { rowCount } = await pool.query(
    `UPDATE ${TB}
        SET is_active = FALSE,
            updated_at = NOW()
      WHERE customer_id = $1`,
    [cid]
  );

  return {
    ok: rowCount > 0,
    hard: false,
  };
}

/* ──────────────────────────────────────────────────────────

export async function list({
  q,

  business_segment,
  seller_id,
  segment_id,
  limit = 100,
  offset = 0,
  orderBy = 'trade_name',
  order = 'asc',
} = {}) {
  const params = [];
  const where = [];
  const joins = [];

  const orderSafe = [
    'trade_name',
    'city',
    'created_at',
    'updated_at',
    'customer_id',
  ].includes(orderBy)
    ? orderBy
    : 'trade_name';

  const dir =
    String(order).toLowerCase() === 'desc'
      ? 'DESC'
      : 'ASC';

  joins.push(`
    LEFT JOIN public."customerStatus" cs
      ON cs.customer_id = ${ALIAS}.customer_id

    LEFT JOIN public."customerStatusTypes" cst
      ON cst.id = cs.status_type_id
  `);

    LEFT JOIN ${TB_SEG} seg
      ON seg.segment_id = ${ALIAS}.segment_id
  `);

    LEFT JOIN LATERAL (
      SELECT sfc1.seller_id
      FROM public."SellerForCustomer" sfc1
      WHERE sfc1.customer_id = ${ALIAS}.customer_id
        AND (sfc1.active IS NULL OR sfc1.active = TRUE)
      ORDER BY sfc1.created_at DESC
      LIMIT 1
    ) sfc ON TRUE

    LEFT JOIN public.sellers s
      ON s.id = sfc.seller_id
  `);

    LEFT JOIN ${TB_IBGE} ci
      ON ci.customer_id = ${ALIAS}.customer_id
  `);

  joins.push(`
    LEFT JOIN ${TB_DELIVERY} ad
      ON ad.customer_ad_id = ${ALIAS}.customer_ad_id
  `);

  if (q && String(q).trim()) {
    const norm = String(q)
      .toLowerCase()
      .trim();

    params.push(`%${norm}%`);

    where.push(`(
      LOWER(COALESCE(${ALIAS}.trade_name,'')) LIKE $${params.length} OR
      LOWER(COALESCE(${ALIAS}.legal_name,'')) LIKE $${params.length} OR
      COALESCE(${ALIAS}.tax_id,'') LIKE $${params.length}
    )`);
  }

  if (typeof is_active === 'boolean') {
    params.push(is_active);
    where.push(
      `${ALIAS}.is_active = $${params.length}`
    );
  }

  if (Number.isFinite(Number(status_type_id))) {
    params.push(Number(status_type_id));
    where.push(
      `cs.status_type_id = $${params.length}`
    );
  }

  if (Number.isFinite(Number(seller_id))) {
    params.push(Number(seller_id));
    where.push(
      `sfc.seller_id = $${params.length}`
    );
  }

  if (Number.isFinite(Number(segment_id))) {
    params.push(Number(segment_id));
    where.push(
      `${ALIAS}.segment_id = $${params.length}`
    );
  }

  const whereSql =
    where.length
      ? `WHERE ${where.join(' AND ')}`
      : '';

  const lim =
    toNumOrNull(limit) ??
    100;

  const off =
    toNumOrNull(offset) ??
    0;

  params.push(lim);
  const limIdx = params.length;

  params.push(off);
  const offIdx = params.length;

  const sql = `
    SELECT
      ${ALIAS}.*,

      -- IBGE
      ci.codigo_ibge      AS ibge_city_code,
      ci.source           AS ibge_source,
      ci.confidence       AS ibge_confidence,

      ad.customer_ad_address_complement,
      ad.customer_ad_neighborhood,
      ad.customer_ad_city,
      ad.customer_ad_state,
      ad.active AS customer_ad_active

    FROM ${TB} ${ALIAS}

    ${joins.join('\n')}

    ${whereSql}

    ORDER BY ${ALIAS}."${orderSafe}" ${dir}

    LIMIT $${limIdx}
    OFFSET $${offIdx};
  `;

  const countSql = `
    SELECT COUNT(*)::INT AS total
    FROM ${TB} ${ALIAS}

    ${joins.join('\n')}

    ${whereSql};
  `;

  const countParams =
    params.slice(
      0,
      params.length - 2
    );

  const [rowsRes, totalRes] =
    await Promise.all([
      pool.query(sql, params),
      pool.query(countSql, countParams),
    ]);

  return {
    rows: rowsRes.rows.map(mapRow),
    total: totalRes.rows[0]?.total ?? 0,
    limit: lim,
    offset: off,
  };
}

/* ──────────────────────────────────────────────────────────
   SEARCH
────────────────────────────────────────────────────────── */

export async function search({
  q,
  is_active = true,
  limit = 10,
  offset = 0,
} = {}) {
  const params = [];
  const where = [];

  params.push(!!is_active);

  where.push(
    `${ALIAS}.is_active = $${params.length}`
  );

  let likeAll;
  let likeStart;

  if (q && String(q).trim()) {
    const norm = String(q)
      .toLowerCase()
      .trim();

    likeAll = `%${norm}%`;
    likeStart = `${norm}%`;

    params.push(likeAll);
    params.push(likeAll);
    params.push(likeAll);

    where.push(`(
      LOWER(COALESCE(${ALIAS}.trade_name,'')) LIKE $${params.length - 2} OR
      LOWER(COALESCE(${ALIAS}.legal_name,'')) LIKE $${params.length - 1} OR
      COALESCE(${ALIAS}.tax_id,'') LIKE $${params.length}
    )`);
  }

  const whereSql =
    where.length
      ? `WHERE ${where.join(' AND ')}`
      : '';

  let orderSql =
    `ORDER BY LOWER(COALESCE(${ALIAS}.trade_name,'')) ASC`;

  if (q && likeStart) {
    params.push(likeStart);

    orderSql = `
      ORDER BY
        CASE
          WHEN LOWER(COALESCE(${ALIAS}.trade_name,'')) LIKE $${params.length}
            THEN 0
          ELSE 1
        END,
        LOWER(COALESCE(${ALIAS}.trade_name,'')) ASC
    `;
  }

  const lim =
    Number.isFinite(+limit)
      ? +limit
      : 10;

  const off =
    Number.isFinite(+offset)
      ? +offset
      : 0;

  params.push(lim);
  const limIdx = params.length;

  params.push(off);
  const offIdx = params.length;

  const sql = `
    SELECT
      ${ALIAS}.customer_id,
      ${ALIAS}.customer_type,
      ${ALIAS}.trade_name,
      ${ALIAS}.tax_id,
      ${ALIAS}.city,
      ${ALIAS}.state,
      ${ALIAS}.email,
      ${ALIAS}.phone_number,
      ${ALIAS}.mobile_number,

      ${ALIAS}.segment_id,
      seg.segment_name AS segment_name,

      cs.status_type_id,
      COALESCE(cst.name_pt, cst.internal_code) AS status_label

    FROM ${TB} ${ALIAS}

    LEFT JOIN public."customerStatus" cs
      ON cs.customer_id = ${ALIAS}.customer_id

    LEFT JOIN public."customerStatusTypes" cst
      ON cst.id = cs.status_type_id

    LEFT JOIN ${TB_SEG} seg
      ON seg.segment_id = ${ALIAS}.segment_id

    ${whereSql}

    ${orderSql}

    LIMIT $${limIdx}
    OFFSET $${offIdx};
  `;

  const { rows } =
    await pool.query(
      sql,
      params
    );

  return {
    rows: rows.map(mapSearchRow),
    limit: lim,
    offset: off,
  };
}

  if (Number.isFinite(Number(status_type_id))) {
    params.push(Number(status_type_id));

    where.push(
      `cs.status_type_id = $${params.length}`
    );
  }

  if (Number.isFinite(Number(seller_id))) {
    params.push(Number(seller_id));

    where.push(
      `sfc.seller_id = $${params.length}`
    );
  }

  if (Number.isFinite(Number(segment_id))) {
    params.push(Number(segment_id));

    where.push(
      `${ALIAS}.segment_id = $${params.length}`
    );
  }

  const whereSql =
    where.length
      ? `WHERE ${where.join(' AND ')}`
      : '';

  const sql = `
    SELECT
      ${ALIAS}.*,

      ci.codigo_ibge      AS ibge_city_code,
      ci.source           AS ibge_source,
      ci.confidence       AS ibge_confidence,

      cs.status_type_id,
      cst.name_pt         AS status_name_pt,
      cst.internal_code   AS status_internal_code,
      cst.color_hex       AS status_color_hex,
      COALESCE(cst.name_pt, cst.internal_code) AS status_label,

      seg.segment_code    AS segment_code,
      seg.segment_name    AS segment_name,

      sfc.seller_id       AS seller_id,
      COALESCE(s.nickname, s.name) AS seller_name,

      ad.customer_ad_street_type,
      ad.customer_ad_street_name,
      ad.customer_ad_street_number,
      ad.customer_ad_address_complement,
      ad.customer_ad_neighborhood,
      ad.customer_ad_city,
      ad.customer_ad_state,
      ad.active AS customer_ad_active

    FROM ${TB} ${ALIAS}

    ${joins.join('\n')}

    ${whereSql}

    ORDER BY ${ALIAS}."${orderSafe}" ${dir};
  `;

  const { rows } =
    await pool.query(
      sql,
      params
    );

  return rows.map(mapRow);
}
