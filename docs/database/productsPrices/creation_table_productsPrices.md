BEGIN;

CREATE TABLE public.".productsPrices"
(
    product_price_id INTEGER GENERATED BY DEFAULT AS IDENTITY,

    product_code VARCHAR(50) NOT NULL,

    price_id INTEGER NOT NULL,

    sale_price NUMERIC(10,2),

    markup_percent NUMERIC(5,2),

    gross_profit_pct NUMERIC(10,2),

    created_at TIMESTAMP WITHOUT TIME ZONE
        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP WITHOUT TIME ZONE
        NOT NULL DEFAULT CURRENT_TIMESTAMP,

    user_id INTEGER,

    CONSTRAINT ".productsPrices_pkey"
        PRIMARY KEY (product_price_id),

    CONSTRAINT ".productsPrices_product_price_uk"
        UNIQUE (product_code, price_id)
);



ALTER TABLE public.".productsPrices"
ADD CONSTRAINT ".productsPrices_product_code_fkey"
FOREIGN KEY (product_code)
REFERENCES public.products(product_code)
ON UPDATE CASCADE
ON DELETE RESTRICT;



ALTER TABLE public.".productsPrices"
ADD CONSTRAINT ".productsPrices_price_id_fkey"
FOREIGN KEY (price_id)
REFERENCES public.prices(price_id)
ON UPDATE CASCADE
ON DELETE RESTRICT;



ALTER TABLE public.".productsPrices"
ADD CONSTRAINT ".productsPrices_user_id_fkey"
FOREIGN KEY (user_id)
REFERENCES public.users(user_id)
ON UPDATE CASCADE
ON DELETE SET NULL;
