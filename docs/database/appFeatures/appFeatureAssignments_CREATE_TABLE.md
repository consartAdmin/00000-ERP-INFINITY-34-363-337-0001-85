```SQL

BEGIN;

CREATE TABLE public."appFeatureAssignments" (
    feature_assignment_id BIGINT
        GENERATED BY DEFAULT AS IDENTITY,

    feature_id BIGINT NOT NULL,
    feature_option_id BIGINT,

    scope_type VARCHAR(30) NOT NULL,
    scope_id BIGINT NOT NULL,

    custom_value JSONB,
    is_enabled BOOLEAN NOT NULL DEFAULT TRUE,

    starts_at TIMESTAMPTZ,
    ends_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT "appFeatureAssignments_pkey"
        PRIMARY KEY (feature_assignment_id),

    CONSTRAINT "uq_appFeatureAssignments_scope"
        UNIQUE (scope_type, scope_id, feature_id),

    CONSTRAINT "ck_appFeatureAssignments_period"
        CHECK (
            ends_at IS NULL
            OR starts_at IS NULL
            OR ends_at > starts_at
        ),

    CONSTRAINT "ck_appFeatureAssignments_scope_type"
        CHECK (scope_type::TEXT = UPPER(scope_type::TEXT)),

    CONSTRAINT "fk_appFeatureAssignments_feature"
        FOREIGN KEY (feature_id)
        REFERENCES public."appFeatures" (feature_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT "fk_appFeatureAssignments_option"
        FOREIGN KEY (feature_id, feature_option_id)
        REFERENCES public."appFeatureOptions" (
            feature_id,
            feature_option_id
        )
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);

CREATE INDEX "idx_appFeatureAssignments_lookup"
    ON public."appFeatureAssignments"
    USING BTREE (scope_type, scope_id, is_enabled);

CREATE TRIGGER "trg_appFeatureAssignments_updatedAt"
    BEFORE UPDATE ON public."appFeatureAssignments"
    FOR EACH ROW
    EXECUTE FUNCTION public."setAppFeatureUpdatedAt"();

COMMIT;
```
