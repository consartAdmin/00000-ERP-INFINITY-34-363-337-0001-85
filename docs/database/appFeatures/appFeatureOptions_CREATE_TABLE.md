```SQL

BEGIN;

CREATE TABLE public."appFeatureOptions" (
    feature_option_id BIGINT
        GENERATED BY DEFAULT AS IDENTITY,

    feature_id BIGINT NOT NULL,
    option_code VARCHAR(100) NOT NULL,
    option_name VARCHAR(150) NOT NULL,
    description TEXT,

    option_config JSONB NOT NULL DEFAULT '{}'::JSONB,
    sort_order INTEGER NOT NULL DEFAULT 0,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT "appFeatureOptions_pkey"
        PRIMARY KEY (feature_option_id),

    CONSTRAINT "uq_appFeatureOptions_code"
        UNIQUE (feature_id, option_code),

    CONSTRAINT "uq_appFeatureOptions_feature_option"
        UNIQUE (feature_id, feature_option_id),

    CONSTRAINT "ck_appFeatureOptions_code"
        CHECK (option_code::TEXT = UPPER(option_code::TEXT)),

    CONSTRAINT "fk_appFeatureOptions_feature"
        FOREIGN KEY (feature_id)
        REFERENCES public."appFeatures" (feature_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

CREATE INDEX "idx_appFeatureOptions_active"
    ON public."appFeatureOptions"
    USING BTREE (feature_id, is_active, sort_order);

CREATE TRIGGER "trg_appFeatureOptions_updatedAt"
    BEFORE UPDATE ON public."appFeatureOptions"
    FOR EACH ROW
    EXECUTE FUNCTION public."setAppFeatureUpdatedAt"();

COMMIT;
```
