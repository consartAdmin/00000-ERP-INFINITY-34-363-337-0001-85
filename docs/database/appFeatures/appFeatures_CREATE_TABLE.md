-- ============================================================
-- TABLE: public."appFeatures"
-- CONSART ERP — Configuração de funcionalidades
-- ============================================================

CREATE TABLE public."appFeatures" (
    feature_id BIGINT GENERATED BY DEFAULT AS IDENTITY,

    feature_code VARCHAR(100) NOT NULL,
    feature_name VARCHAR(150) NOT NULL,
    description TEXT,

    value_type VARCHAR(20) NOT NULL DEFAULT 'OPTION',
    default_value JSONB,

    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT "appFeatures_pkey"
        PRIMARY KEY (feature_id),

    CONSTRAINT "uq_appFeatures_code"
        UNIQUE (feature_code),

    CONSTRAINT "ck_appFeatures_code"
        CHECK (feature_code = UPPER(feature_code)),

    CONSTRAINT "ck_appFeatures_value_type"
        CHECK (
            value_type IN (
                'BOOLEAN',
                'OPTION',
                'NUMBER',
                'TEXT',
                'JSON'
            )
        )
);

-- ============================================================
-- FUNCTION: Atualiza automaticamente updated_at
-- ============================================================

CREATE OR REPLACE FUNCTION public."setAppFeatureUpdatedAt"()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.updated_at := NOW();
    RETURN NEW;
END;
$$;

-- ============================================================
-- TRIGGER
-- ============================================================

CREATE TRIGGER "trg_appFeatures_updatedAt"
BEFORE UPDATE ON public."appFeatures"
FOR EACH ROW
EXECUTE FUNCTION public."setAppFeatureUpdatedAt"();
